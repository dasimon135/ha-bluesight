# BlueSight v1.2 — Proxy Health Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Surface ESPHome Bluetooth-proxy health (offline, stalled/zombie, reboot-storm) as BlueSight incidents + per-proxy entities, sourced entirely from the `habluetooth` scanner registry (zero user config).

**Architecture:** Reuse the v1 pattern — pure HA-free detectors + a thin adapter over the habluetooth manager + the existing `FailureWindow` + coordinator + incident model. Proxy incidents append to the existing `incidents` list so the global binary_sensor, notifications, and dashboard pick them up unchanged.

**Tech Stack:** Python 3.14, Home Assistant custom integration, `habluetooth` scanner registry, pytest. English only.

```python
def detect_offline_proxies(
    proxies: list[ProxyHealth], known_sources: set[str]
) -> list[Incident]:
    """A source we have seen online before, now absent from the scanners."""
    online = {p.source for p in proxies if p.online}
    return [
        Incident(IncidentKind.PROXY_OFFLINE, src, [src],
                 detail="Proxy offline (no longer a registered scanner)")
        for src in sorted(known_sources - online)
    ]
```

```python
def detect_stalled_proxies(
    proxies: list[ProxyHealth], threshold_s: float
) -> list[Incident]:
    """Online scanner that has not seen any advertisement for too long."""
    return [
        Incident(IncidentKind.PROXY_STALLED, p.source, [p.source],
                 detail=f"Online but no BLE advertisement seen for "
                        f"{int(p.seconds_since_detection)}s")
        for p in proxies
        if p.online and p.seconds_since_detection > threshold_s
    ]
```

```python
def detect_reboot_storm(source: str, window: FailureWindow) -> Incident | None:
    count = window.count(source)
    if count >= window.threshold:
        return Incident(IncidentKind.PROXY_REBOOT_STORM, source, [source],
                        detail=f"{count} proxy reboots in "
                               f"{int(window.window_s)}s")
    return None
```

**Step 3 — implement in `adapter.py`:**
```python
def current_proxy_health(manager: Any) -> list[ProxyHealth]:
    scanners = manager.async_current_scanners() or []
    return [
        ProxyHealth(
            source=s.source,
            name=getattr(s, "name", s.source) or s.source,
            connectable=bool(getattr(s, "connectable", False)),
            online=True,
            seconds_since_detection=float(s.time_since_last_detection()),
            device_count=len(s.discovered_devices),
        )
        for s in scanners
    ]

class ScannerAdapter:
    """Subscribe to habluetooth scanner registration events. Fires on_change on
    every event and on_removed(source) when a scanner is removed (reboot signal)."""

    def __init__(self, manager: Any, on_change: Callable[[], None],
                 on_removed: Callable[[str], None]) -> None:
        self._manager = manager
        self._on_change = on_change
        self._on_removed = on_removed
        self._unsub: Callable[[], None] | None = None

    def start(self) -> None:
        if self._unsub is not None:
            return
        self._unsub = self._manager.async_register_scanner_registration_callback(
            self._handle, None)

    def _handle(self, registration: Any) -> None:
        event = getattr(registration.event, "value", registration.event)
        if event == "removed":
            self._on_removed(registration.scanner.source)
        self._on_change()

    def stop(self) -> None:
        if self._unsub is not None:
            self._unsub()
            self._unsub = None
```

```python
@dataclass(frozen=True, slots=True)
class BleTriageData:
    proxies: list[ProxySlots] = field(default_factory=list)
    incidents: list[Incident] = field(default_factory=list)
    proxies_health: list[ProxyHealth] = field(default_factory=list)

def build_triage_data(
    proxies, availability, storm_window,
    *, proxies_health=None, known_sources=None,
    reboot_window=None, stalled_threshold_s=180.0,
) -> BleTriageData:
    proxies_health = proxies_health or []
    known_sources = known_sources or set()
    incidents = []
    incidents += detect_deadlocks(proxies)
    incidents += detect_ghost_slots(proxies, availability)
    for addr in storm_window.addresses():
        if (inc := detect_storm(addr, storm_window)) is not None:
            incidents.append(inc)
    # proxy health
    incidents += detect_offline_proxies(proxies_health, known_sources)
    incidents += detect_stalled_proxies(proxies_health, stalled_threshold_s)
    if reboot_window is not None:
        for src in reboot_window.addresses():
            if (inc := detect_reboot_storm(src, reboot_window)) is not None:
                incidents.append(inc)
    return BleTriageData(proxies=proxies, incidents=incidents,
                         proxies_health=proxies_health)
```
